/**
 * ==========================================
 * 第2層：計算エンジン
 * ==========================================
 */
function calcSalaryDeduction(revYen, minDed) {
  if (revYen <= 0) return 0;
  let std = 0;
  if (revYen <= 1625000) std = 550000;
  else if (revYen <= 1800000) std = revYen * 0.4 - 100000;
  else if (revYen <= 3600000) std = revYen * 0.3 + 80000;
  else if (revYen <= 6600000) std = revYen * 0.2 + 440000;
  else if (revYen <= 8500000) std = revYen * 0.1 + 1100000;
  else std = 1950000;
  return Math.max(std, minDed);
}

function calcProgressiveIncomeTax(taxableYen) {
  if (taxableYen <= 0) return 0;
  const brackets = [
    { limit: 1950000, rate: 0.05, ded: 0 },
    { limit: 3300000, rate: 0.10, ded: 97500 },
    { limit: 6950000, rate: 0.20, ded: 427500 },
    { limit: 9000000, rate: 0.23, ded: 636000 },
    { limit: 18000000, rate: 0.33, ded: 1536000 },
    { limit: 40000000, rate: 0.40, ded: 2796000 },
    { limit: Infinity, rate: 0.45, ded: 4796000 }
  ];
  for (const b of brackets) {
    if (taxableYen <= b.limit) return Math.floor(taxableYen * b.rate - b.ded);
  }
  return 0;
}

function evaluatePattern(p, yearKey) {
  const master = TAX_MASTER[yearKey];
  const sRev = p.salaryRev * 10000;
  const bBefore = p.bizIncomeBeforeBlue * 10000;
  const bDed = p.blueDeduction * 10000;
  const stRev = p.stockRev * 10000;
  const stCost = p.stockCost * 10000;
  const divInc = p.divIncome * 10000;
  const socIns = p.socialIns * 10000;
  const othDed = p.otherDeduction * 10000;

  // 1. 各種所得の計算
  const salaryDed = calcSalaryDeduction(sRev, master.salaryMinDeduction);
  const salaryIncome = Math.max(0, sRev - salaryDed);
  const bizIncome = Math.max(0, bBefore - bDed);
  const stockGain = Math.max(0, stRev - stCost);

  // 申告する所得の振り分け
  const declaredStock = (p.stockMethod === "separate") ? stockGain : 0;
  const declaredDivComp = (p.divMethod === "comprehensive") ? divInc : 0;
  const declaredDivSep = (p.divMethod === "separate") ? divInc : 0;

  // 合計所得金額（基礎控除の判定用）
  const totalDeclaredIncome = salaryIncome + bizIncome + declaredStock + declaredDivComp + declaredDivSep;

  // 2. 基礎控除の判定
  let itBasicDed = 0;
  for (const row of master.basicDeductionTable) {
    if (totalDeclaredIncome <= row.maxIncome) {
      itBasicDed = row.amount;
      break;
    }
  }
  const resBasicDed = (totalDeclaredIncome <= 24000000) ? master.residentBasicDeduction : 0;

  const totalItDed = itBasicDed + socIns + othDed;
  const totalResDed = resBasicDed + socIns + othDed;

  // 3. 課税所得の算出（総合課税から先に引き、余りを分離課税から引く）
  const compIncome = salaryIncome + bizIncome + declaredDivComp;
  const sepIncome = declaredStock + declaredDivSep;

  // 所得税の課税所得
  const itTaxableComp = Math.floor(Math.max(0, compIncome - totalItDed) / 1000) * 1000;
  const itUnusedDed = Math.max(0, totalItDed - compIncome);
  const itTaxableSep = Math.floor(Math.max(0, sepIncome - itUnusedDed) / 1000) * 1000;

  // 住民税の課税所得
  const resTaxableComp = Math.floor(Math.max(0, compIncome - totalResDed) / 1000) * 1000;
  const resUnusedDed = Math.max(0, totalResDed - compIncome);
  const resTaxableSep = Math.floor(Math.max(0, sepIncome - resUnusedDed) / 1000) * 1000;

  // 4. 所得税の計算（確定申告分）
  let itBaseComp = calcProgressiveIncomeTax(itTaxableComp);
  // 配当控除（課税総所得金額1000万円以下部分は10%、超える部分は5%）
  let itDivCredit = 0;
  if (declaredDivComp > 0 && itTaxableComp > 0) {
    const eligibleDiv = Math.min(declaredDivComp, itTaxableComp);
    const totalTaxable = itTaxableComp + itTaxableSep;
    if (totalTaxable <= 10000000) {
      itDivCredit = Math.floor(eligibleDiv * 0.10);
    } else {
      const under10M = Math.max(0, 10000000 - (totalTaxable - eligibleDiv));
      const over10M = eligibleDiv - under10M;
      itDivCredit = Math.floor(under10M * 0.10 + over10M * 0.05);
    }
  }
  const itCompAfterCredit = Math.max(0, itBaseComp - itDivCredit);
  const itSepTax = Math.floor(itTaxableSep * 0.15);
  const itDeclaredTotal = Math.floor((itCompAfterCredit + itSepTax) * (1 + master.surtaxRate));

  // 申告不要を選んだ分の源泉徴収税（所得税15.315%）
  const withheldItStock = (p.stockMethod === "no_declare") ? Math.floor(stockGain * 0.15315) : 0;
  const withheldItDiv = (p.divMethod === "no_declare") ? Math.floor(divInc * 0.15315) : 0;
  const finalIncomeTax = itDeclaredTotal + withheldItStock + withheldItDiv;

  // 5. 住民税の計算
  let resCompTax = Math.floor(resTaxableComp * 0.10);
  let resDivCredit = 0;
  if (declaredDivComp > 0 && resTaxableComp > 0) {
    const eligibleResDiv = Math.min(declaredDivComp, resTaxableComp);
    const totalResTaxable = resTaxableComp + resTaxableSep;
    if (totalResTaxable <= 10000000) {
      resDivCredit = Math.floor(eligibleResDiv * 0.028);
    } else {
      const under10M = Math.max(0, 10000000 - (totalResTaxable - eligibleResDiv));
      const over10M = eligibleResDiv - under10M;
      resDivCredit = Math.floor(under10M * 0.028 + over10M * 0.014);
    }
  }
  const resSepTax = Math.floor(resTaxableSep * 0.05);
  const adjCredit = (resTaxableComp + resTaxableSep > 0) ? 2500 : 0;
  const resIncomeLevy = Math.floor(Math.max(0, resCompTax - resDivCredit + resSepTax - adjCredit) / 100) * 100;
  const perCapita = (totalDeclaredIncome > 450000) ? master.residentPerCapita : 0;
  const withheldResStock = (p.stockMethod === "no_declare") ? Math.floor(stockGain * 0.05) : 0;
  const withheldResDiv = (p.divMethod === "no_declare") ? Math.floor(divInc * 0.05) : 0;
  const finalResidentTax = resIncomeLevy + perCapita + withheldResStock + withheldResDiv;

  // 源泉徴収本来の総額（比較用）
  const originalWithheld = Math.floor((stockGain + divInc) * 0.20315);
  const totalTax = finalIncomeTax + finalResidentTax;
  const taxSaved = originalWithheld - totalTax;

  // 手取り計算
  const totalCashIn = sRev + bBefore + stRev + divInc;
  const totalProfitIn = sRev + bBefore + stockGain + divInc;
  const netCash = totalCashIn - socIns - othDed - totalTax;
  const netProfit = totalProfitIn - socIns - othDed - totalTax;

  return {
    salaryIncome, bizIncome, stockGain, divInc,
    itBasicDed, totalItDed, totalResDed,
    finalIncomeTax, finalResidentTax, totalTax,
    originalWithheld, taxSaved, netCash, netProfit
  };
}

        <span class="unit">万円</span>
      </div>
    </div>
  `;
}

function fmtYen(n) {
  return n.toLocaleString("ja-JP") + "円";
}

function renderResults(key, r) {
  const box = document.getElementById("res_" + key);
  const savedLabel = r.taxSaved >= 0
    ? `<span style="color:#059669;font-weight:700;">▲${fmtYen(r.taxSaved)} 節税</span>`
    : `<span style="color:#dc2626;font-weight:700;">+${fmtYen(Math.abs(r.taxSaved))} 増税</span>`;

  box.innerHTML = `
    <div class="res-row"><span>株式譲渡益 / 配当所得</span><span>${fmtYen(r.stockGain)} / ${fmtYen(r.divInc)}</span></div>
    <div class="res-row"><span>判定された所得税の基礎控除</span><span>${fmtYen(r.itBasicDed)}</span></div>
    <div class="res-row"><span>所得税（復興特別所得税含）</span><span>${fmtYen(r.finalIncomeTax)}</span></div>
    <div class="res-row"><span>住民税（均等割5,000円含）</span><span>${fmtYen(r.finalResidentTax)}</span></div>
    <div class="res-row tax-total"><span>年間税額 合計</span><span>${fmtYen(r.totalTax)}</span></div>
    <div class="res-row"><span>源泉徴収額(${fmtYen(r.originalWithheld)})との差</span><span>${savedLabel}</span></div>
    <div class="res-row highlight"><span>手取り（純利益ベース）</span><span>${fmtYen(r.netProfit)}</span></div>
    <div class="res-row highlight"><span>手取り（総入金キャッシュ）</span><span>${fmtYen(r.netCash)}</span></div>
  `;
  document.getElementById("badge" + key).textContent = `税額 ${Math.round(r.totalTax/10000*10)/10}万円`;
}

function calculateAll() {
  const yearKey = document.getElementById("taxYear").value;
  const resA = evaluatePattern(appState.A, yearKey);
  const resB = evaluatePattern(appState.B, yearKey);

  renderResults("A", resA);
  renderResults("B", resB);

  // フッター更新
  document.getElementById("footA").textContent = `${Math.round(resA.totalTax/10000*10)/10}万 / ${Math.round(resA.netCash/10000*10)/10}万円`;
  document.getElementById("footB").textContent = `${Math.round(resB.totalTax/10000*10)/10}万 / ${Math.round(resB.netCash/10000*10)/10}万円`;
  
  const diff = resA.netCash - resB.netCash;
  const sign = diff > 0 ? "+" : "";
  document.getElementById("footDiff").textContent = `${sign}${fmtYen(diff)}`;

  localStorage.setItem("taxSimState_v1", JSON.stringify(appState));
}

function updateVal(key, field, val) {
  appState[key][field] = Math.max(0, Number(val) || 0);
  calculateAll();
}

function stepVal(key, field, delta) {
  const next = Math.max(0, (Number(appState[key][field]) || 0) + delta);
  appState[key][field] = next;
  document.getElementById(`${key}_${field}`).value = next;
  calculateAll();
}

function updateRadio(key, field, val) {
  appState[key][field] = val;
  calculateAll();
}

function copyAtoB() {
  appState.B = JSON.parse(JSON.stringify(appState.A));
  initForms();
}

function resetPreset() {
  appState = JSON.parse(JSON.stringify(DEFAULT_STATE));
  initForms();
}

function switchTab(target) {
  const pA = document.getElementById("panelA");
  const pB = document.getElementById("panelB");
  const tA = document.getElementById("tabA");
  const tB = document.getElementById("tabB");
  if (target === "A") {
    pA.style.display = "block";
    pB.style.display = "none";
    tA.className = "tab-btn active-a";
    tB.className = "tab-btn";
  } else {
    pA.style.display = "none";
    pB.style.display = "block";
    tA.className = "tab-btn";
    tB.className = "tab-btn active-b";
  }
}

function initForms() {
  document.getElementById("formA").innerHTML = buildFormHTML("A");
  document.getElementById("formB").innerHTML = buildFormHTML("B");
  calculateAll();
}

window.addEventListener("DOMContentLoaded", initForms);
